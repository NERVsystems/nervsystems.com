% Checklist item with checkbox
\newcommand{\checkitem}[1]{%
  \item[\textcolor{tactical-accent}{$\square$}] #1
}

% Completed checklist item
\newcommand{\checkeditem}[1]{%
  \item[\textcolor{tactical-accent}{$\boxtimes$}] #1
}

% Priority levels
\newcommand{\priority}[1]{%
  \textcolor{tactical-accent}{\texttt{[P#1]}}
}

% Tactical checklist environment
\newenvironment{tacticalchecklist}{%
  \begin{mdframed}[
    linecolor=tactical-accent,
    linewidth=1pt,
    backgroundcolor=tactical-surface!30,
    leftmargin=0pt,
    rightmargin=0pt,
    innerleftmargin=10pt,
    innerrightmargin=10pt,
    innertopmargin=10pt,
    innerbottommargin=10pt,
    skipabove=10pt,
    skipbelow=10pt
  ]
  \begin{itemize}[leftmargin=*,labelsep=10pt]
}{%
  \end{itemize}
  \end{mdframed}
}

% Phase section (for multi-phase guides)
\newcounter{phase}
\newcommand{\phase}[1]{%
  \stepcounter{phase}
  \section*{\textcolor{tactical-accent}{\texttt{[PHASE \arabic{phase}]}} #1}
  \addcontentsline{toc}{section}{\protect\numberline{Phase \arabic{phase}}#1}
}

% Timeline item
\newcommand{\timeline}[2]{%
  \noindent\textcolor{tactical-accent}{\texttt{[\,#1\,]}} \textbf{#2}\\[0.5em]
}

% Step-by-step procedure
\newcounter{step}[section]
\newenvironment{procedure}{%
  \setcounter{step}{0}
  \begin{enumerate}[
    label=\textcolor{tactical-accent}{\texttt{[\arabic*]}},
    leftmargin=*,
    labelsep=10pt
  ]
}{%
  \end{enumerate}
}

% Sub-steps
\newenvironment{substeps}{%
  \begin{enumerate}[
    label=\textcolor{tactical-dim}{\texttt{\alph*)}},
    leftmargin=*,
    labelsep=8pt
  ]
}{%
  \end{enumerate}
}
