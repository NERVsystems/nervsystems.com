% HUD corner brackets
\newcommand{\hudcorners}[1]{%
  \begin{tikzpicture}[remember picture, overlay]
    \draw[tactical-accent, line width=1pt]
      ([xshift=5pt,yshift=-5pt]current page.north west) --
      ([xshift=5pt,yshift=-20pt]current page.north west);
    \draw[tactical-accent, line width=1pt]
      ([xshift=5pt,yshift=-5pt]current page.north west) --
      ([xshift=20pt,yshift=-5pt]current page.north west);

    \draw[tactical-accent, line width=1pt]
      ([xshift=-5pt,yshift=-5pt]current page.north east) --
      ([xshift=-5pt,yshift=-20pt]current page.north east);
    \draw[tactical-accent, line width=1pt]
      ([xshift=-5pt,yshift=-5pt]current page.north east) --
      ([xshift=-20pt,yshift=-5pt]current page.north east);
  \end{tikzpicture}%
}
